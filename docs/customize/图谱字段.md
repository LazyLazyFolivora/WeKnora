## Company
id              String   // 统一社会信用代码（国内）/ LEI码（国际）
name            String   // 标准名称（中文）
name_en         String   // 英文名
aliases         String[] // 别名列表（简称/曾用名/股票代码）
types           String[] // 多值数组，支持多角色
                         // ["biotech","pharma","cro","cmo",
                         //  "investment","regulator","academic"]
country         String
founded_year    Integer
stage           String   // "早期研发" / "临床阶段" / "商业化阶段"（人工标注）
primary_focus   String   // 主要治疗领域（人工标注，如"肿瘤"）

### 可选（非上市公司为空）
stock_code      String
exchange        String   // "A股" / "港股" / "NASDAQ" / "NYSE"
market_cap      Float    // 当前市值快照（USD）
market_cap_date String   // 市值更新日期


## Drug
id                  String   // DrugBank ID / NMPA批准文号 / 内部ID
name                String   // 通用名（INN名优先）
aliases             String[] // 商品名 / 研发代号 / 曾用名
drug_class          String   // "化学药品" / "生物制品" / "中药"
modality            String   // "小分子" / "单抗" / "双抗" / "多抗" /
                             // "ADC" / "多肽" / "融合蛋白" / "纳米抗体" /
                             // "细胞治疗" / "基因治疗" /
                             // "核酸药物" / "反义寡核苷酸" / "小干扰RNA" /
                             // "放射性药物" / "疫苗" / "溶瘤病毒"
innovation_type     String   // "创新药" / "仿制药" / "改良型新药" / "生物类似药"
innovation_labels   String[] // ["first-in-class","best-in-class","me-too","me-better"]
global_status       String   // "Active" / "Discontinued" / "Unknown"

### 按地区拆平（Neo4j不支持嵌套JSON）
phase_cn            String   // "临床前"/"IND申报"/"I期"/"II期"/"III期"/
phase_us            String   // "NDA申报"/"已上市"/"终止"/"暂停"
phase_eu            String
phase_jp            String

### 化学标识
cas_number          String   // 小分子用，实体消歧

### 专利
patent_expiry_cn    String   // 中国化合物专利到期日
patent_expiry_us    String   // 美国化合物专利到期日


## Target
id                  String   // UniProt ID / HGNC ID
name                String   // 官方基因名/蛋白名（如 CLDN18.2）
gene_symbol         String   // 基因符号（如 CLDN18）
aliases             String[] // 别名/曾用名
target_type         String   // "膜蛋白" / "GPCR" / "酶" / "转录因子" /
                             // "离子通道" / "免疫检查点" / "表面抗原" / "RNA"
target_class        String   // "成熟靶点" / "新兴靶点" / "前沿靶点"
druggability_score  Float    // 成药性评分 0-1
validation_level    String   // "临床验证" / "临床前验证" / "预测"
species             String   // "human" / "mouse" / "rat"（新增）


## Indication
id              String   // MeSH ID / ICD-11编码
name            String
icd_code        String
category        String   // "肿瘤"/"心血管"/"神经"/"内分泌"/"感染"/"免疫"/"罕见病"/"消化"
is_rare         Boolean  // 是否罕见病


## ClinicalTrial
id              String
trial_id        String   // NCT / CTR编号
phase           String   // "I"/"II"/"III"/"IV"/"I/II"/"II/III"
status          String   // "招募中"/"进行中"/"完成"/"终止"/"暂停"
start_date      String
end_date        String
region          String[] // ["中国","美国","欧洲"]
primary_endpoint String
sample_size     Integer
blinding        String   // "开放"/"单盲"/"双盲"
has_result      Boolean  // 是否有结果数据（详细结果存关系型数据库）


## DealEvent
id                  String
deal_category       String   // "融资" / "许可" / "并购" / "战略合作"
deal_sub_types      String[] // 融资：["天使"/"A轮"/"B轮"/"C轮"/"IPO"/"定增"]
                             // 许可：["License-in"/"License-out"/"共同开发"]
                             // 并购：["全资收购"/"部分收购"/"资产收购"]
upfront_usd         Float    // 首付款（美元）
milestone_usd       Float    // 里程碑总额（美元）
total_amount_usd    Float    // 总对价（美元）
currency_orig       String   // 原始货币（如"CNY"）
amount_orig         Float    // 原始金额
exchange_rate_date  String   // 汇率取值日期
rights_region       String[] // ["美国","欧洲","日本","其他"]
exclusivity_type    String   // "独家" / "非独家"
date                String
status              String   // "达成"/"终止"/"传闻"/"修订"
termination_reason  String   // 终止原因（可选）


## ApprovalEvent
id                  String
application_type    String   // "NDA"/"BLA"/"IND"/"sNDA"/"sBLA"/"MAA"
approval_type       String   // "常规批准"/"附条件批准"/"紧急使用授权"（新增）
designations        String[] // ["突破性疗法","优先审评","孤儿药","快速通道"]
approval_number     String   // 批准文号（如国药准字）
status              String   // "申报中"/"审评中"/"获批"/"拒绝"/"撤回"
submit_date         String
approval_date       String
indication_text     String   // 申报适应症文本
conditions          String[] // 附加条件（如"需开展上市后研究"）


## Policy
id              String
title           String
policy_type     String   // "集采"/"医保目录"/"审批改革"/
                         // "临床指南"/"价格管控"/"鼓励政策"/"监管规范"
issue_date      String
effective_date  String
doc_number      String   // 文号
affected_regions String[] // ["全国","北京","上海"]


## TCMFormula
id              String
name            String
aliases         String[]
formula_type    String   // "中成药"/"经典方"/"提取物"/"单味药"
standard        String   // 药典标准
functions       String   // 功效描述（中医语境，保留）


## Compound
id                  String
name                String
cas_number          String
molecular_formula   String
compound_type       String   // "生物碱"/"黄酮"/"皂苷"/"多糖"/"萜类"


## Pathway
id              String   // KEGG pathway ID（如 hsa04151）
name            String
aliases         String[]
pathway_type    String   // "信号通路"/"代谢通路"/"免疫通路"


## DevelopmentProject
id              String   // 内部ID
name            String   // 项目名称（如"A公司与B公司SYSA1801联合开发项目"）
start_date      String   // 合作开始日期
end_date        String   // 合作终止日期（可选）
status          String   // "进行中" / "完成" / "终止"


## DealItem
id              String   // 内部ID
rights_regions  String[] // 该条款对应的权益地区（如["美国","欧洲","日本"]）
item_type       String   // "药物权益" / "技术平台权益"


## TrialIndication
id              String   // 内部ID
phase           String   // 该适应症对应的试验阶段（"I"/"II"/"III"）
status          String   // 该适应症的试验状态（"进行中"/"完成"/"终止"）
orr             Float    // 客观缓解率（可选，有结果时填写）
pfs_months      Float    // 无进展生存期（可选）
os_months       Float    // 总生存期（可选）
result_summary  String   // 结果摘要文本（可选）
